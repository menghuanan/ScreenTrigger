```python
class PerformanceMonitor:
    def __init__(self, operation: str, max_time: float):
        self.operation = operation
```

```python
class ImageCache:
    def __init__(self, max_size_mb: int = 100):
        self.cache = {}
        self.max_size = max_size_mb * 1024 * 1024
        self.current_size = 0
    
    def add(self, key: str, image: np.ndarray):
        size = image.nbytes
        
        # 超过限制时清理最久未使用的
        while self.current_size + size > self.max_size:
            self._evict_oldest()
        
        self.cache[key] = (image, time.time())
        self.current_size += size
    
    def _evict_oldest(self):
        if not self.cache:
            return
        
        oldest_key = min(self.cache, key=lambda k: self.cache[k][1])
        image, _ = self.cache.pop(oldest_key)
        self.current_size -= image.nbytes
```

## 4. 异常处理

### 4.1 系统异常
| 异常类型 | 处理策略 |
|---------|---------|
| 显示器不存在 | 回退到主显示器 |
| 截图失败 | 重试3次，失败后返回错误 |
| OCR引擎崩溃 | 重启引擎，记录日志 |
| 文件不存在 | 返回明确错误信息 |
| 权限不足 | 提示用户以管理员运行 |
| 内存不足 | 清理缓存，降低并发数 |

**异常处理示例**
```python
class CaptureError(Exception):
    pass

def capture_with_retry(region, max_retries=3):
    for attempt in range(max_retries):
        try:
            return capture_region(*region)
        except CaptureError as e:
            if attempt == max_retries - 1:
                raise
            log(f"截图失败，重试 {attempt + 1}/{max_retries}", "warning")
            time.sleep(0.5)
```

```python
def validate_string_input(value: str, field: str,
```

## 6. 并发控制

### 6.1 脚本执行并发

```python
from threading import Semaphore

# 最多同时执行3个脚本
script_execution_semaphore = Semaphore(3)

def execute_script(script_id: str, args: dict):
    if not script_execution_semaphore.acquire(blocking=False):
        raise ConcurrencyError("已有3个脚本在执行，请等待")
    
    try:
        result = _do_execute_script(script_id, args)
        return result
    finally:
        script_execution_semaphore.release()
```

```python
from threading import Lock

file_locks = {}

def get_file_lock(file_path: str) -> Lock:
    if file_path not in file_locks:
        file_locks[file_path] = Lock()
    return file_locks[file_path]

def save_targets_atomic(targets: dict):
    lock = get_file_lock('data/targets.json')
    
    with lock:
        # 先写入临时文件
        temp_path = 'data/targets.json.tmp'
        with open(temp_path, 'w', encoding='utf-8') as f:

        # 原子替换
        os.replace(temp_path, 'data/targets.json')
```

### 7.2 路径安全
```python
def safe_join_path(base: str, *paths: str) -> str:
    """安全的路径拼接，防止路径遍历"""
    result = os.path.normpath(os.path.join(base, *paths))
    
    # 确保结果在base目录内
    if not result.startswith(os.path.normpath(base)):
        raise SecurityError("路径遍历攻击")
    
    return result
```

```python
def sanitize_target_name(name: str) -> str:
    """清理目标名称，移除危险字符"""
    # 移除控制字符
    name = ''.join(c for c in name if c.isprintable())
    
    # 移除文件系统特殊字符
    for char in ['/', '\\', ':', '*', '?', '"', '<', '>', '|']:

```
