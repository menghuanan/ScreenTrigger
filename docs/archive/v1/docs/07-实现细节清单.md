### 1.2 DPI缩放计算
```python
def get_dpi_scale(monitor_handle=None) -> float:
    """获取显示器的DPI缩放比例"""
    try:
        from ctypes import windll, byref, c_uint
        
        if monitor_handle:
            # 获取特定显示器的DPI
            dpi_x = c_uint()
            dpi_y = c_uint()
            windll.shcore.GetDpiForMonitor(
                monitor_handle, 0, byref(dpi_x), byref(dpi_y)
            )
            return dpi_x.value / 96.0
        else:
            # 获取主显示器DPI
            hdc = windll.user32.GetDC(0)
            dpi = windll.gdi32.GetDeviceCaps(hdc, 88)  # LOGPIXELSX
            windll.user32.ReleaseDC(0, hdc)
            return dpi / 96.0
    except:
        return 1.0  # 默认100%缩放
```

### 1.3 坐标转换
```python
class CoordinateConverter:
    def __init__(self, dpi_scale: float):
        self.dpi_scale = dpi_scale
    
    def logical_to_physical(self, x: int, y: int) -> tuple:
        """逻辑坐标 -> 物理坐标"""
        return (
            int(x * self.dpi_scale),
            int(y * self.dpi_scale)
        )
    
    def physical_to_logical(self, x: int, y: int) -> tuple:
        """物理坐标 -> 逻辑坐标"""
        return (
            int(x / self.dpi_scale),
            int(y / self.dpi_scale)
        )
    
    def convert_region(self, region: dict, to_physical: bool) -> dict:
        """转换区域坐标"""
        if to_physical:
            return {
                'x': int(region['x'] * self.dpi_scale),
                'y': int(region['y'] * self.dpi_scale),
                'width': int(region['width'] * self.dpi_scale),
                'height': int(region['height'] * self.dpi_scale)
            }
        else:
            return {
                'x': int(region['x'] / self.dpi_scale),
                'y': int(region['y'] / self.dpi_scale),
                'width': int(region['width'] / self.dpi_scale),
                'height': int(region['height'] / self.dpi_scale)
            }
```

        self.region_x = region['x']
        self.region_y = region['y']
    
    def to_screen(self, relative_x: int, relative_y: int) -> tuple:
        """区域相对坐标 -> 屏幕绝对坐标"""
        return (
            self.region_x + relative_x,
            self.region_y + relative_y

### 3.1 图片格式转换
```python
import cv2
import numpy as np
from PIL import Image
import base64
from io import BytesIO

def base64_to_ndarray(base64_str: str) -> np.ndarray:
    """Base64字符串 -> OpenCV图像"""
    # 解码Base64
    img_data = base64.b64decode(base64_str)
    
    # 转为PIL Image
    pil_img = Image.open(BytesIO(img_data))
    
    # 转为numpy数组（RGB）
    img_array = np.array(pil_img)
    
    # RGB -> BGR（OpenCV格式）
    if len(img_array.shape) == 3:
        img_array = cv2.cvtColor(img_array, cv2.COLOR_RGB2BGR)
    
    return img_array

def ndarray_to_base64(img: np.ndarray, format='PNG') -> str:
    """OpenCV图像 -> Base64字符串"""
    # BGR -> RGB
    if len(img.shape) == 3:
        img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    
    # 转为PIL Image
    pil_img = Image.fromarray(img)
    
    # 编码为字节流
    buffer = BytesIO()
    pil_img.save(buffer, format=format)
    
    # Base64编码
    img_bytes = buffer.getvalue()
    base64_str = base64.b64encode(img_bytes).decode('utf-8')
    
    return base64_str
```

### 3.2 缩略图生成
```python
def generate_thumbnail(img: np.ndarray, size: tuple = (200, 200)) -> np.ndarray:
    """生成缩略图，保持宽高比"""
    h, w = img.shape[:2]
    target_w, target_h = size
    
    # 计算缩放比例
    scale = min(target_w / w, target_h / h)
    
    # 计算新尺寸
    new_w = int(w * scale)
    new_h = int(h * scale)
    
    # 缩放
    thumbnail = cv2.resize(img, (new_w, new_h), interpolation=cv2.INTER_AREA)
    
    # 创建画布（填充黑色）
    canvas = np.zeros((target_h, target_w, 3), dtype=np.uint8)
    
    # 居中放置
    offset_x = (target_w - new_w) // 2
    offset_y = (target_h - new_h) // 2
    canvas[offset_y:offset_y+new_h, offset_x:offset_x+new_w] = thumbnail
    
    return canvas
```

### 3.4 颜色空间转换
```python
def rgb_to_hsv(r: int, g: int, b: int) -> tuple:
    """RGB -> HSV"""
    # 创建1x1的RGB图像
    rgb = np.uint8([[[r, g, b]]])
    
    # 转换为HSV
    hsv = cv2.cvtColor(rgb, cv2.COLOR_RGB2HSV)
    
    h, s, v = hsv[0][0]
    
    # OpenCV的H范围是0-180，需要转换为0-360
    h = int(h * 2)
    s = int(s / 255 * 100)
    v = int(v / 255 * 100)
    
    return (h, s, v)

def hsv_to_rgb(h: float, s: float, v: float) -> tuple:
    """HSV -> RGB"""
    # 转换范围
    h = int(h / 2)  # 360 -> 180
    s = int(s / 100 * 255)
    v = int(v / 100 * 255)
    
    # 创建1x1的HSV图像
    hsv = np.uint8([[[h, s, v]]])
    
    # 转换为RGB
    rgb = cv2.cvtColor(hsv, cv2.COLOR_HSV2RGB)
    
    r, g, b = rgb[0][0]
    
    return (int(r), int(g), int(b))
```

import lupa

class LuaEngine:
    def __init__(self):
        self.lua = lupa.LuaRuntime(unpack_returned_tuples=True)
        self._setup_sandbox()
    
    def _setup_sandbox(self):
        """设置安全的Lua环境"""
        # 禁用危险函数
        self.lua.execute("""
            os = nil
            io = nil
            loadfile = nil
            dofile = nil
            load = nil
            debug = nil
        """)
    
    def register_function(self, name: str, func: callable):
        """注册Python函数到Lua"""
        self.lua.globals()[name] = func
    
    def execute(self, code: str):
        """执行Lua代码"""
        return self.lua.execute(code)
    
    def call_function(self, func_name: str, *args):
        """调用Lua函数"""
        func = self.lua.globals()[func_name]
        return func(*args)

def lua_table_to_dict(lua_table) -> dict:
    """Lua表 -> Python字典"""
    if lua_table is None:
        return {}
    
    result = {}
    for key, value in lua_table.items():
        # 递归转换嵌套表
        if isinstance(value, lupa.lua_type('table')):
            value = lua_table_to_dict(value)
        result[key] = value

def dict_to_lua_table(engine: LuaEngine, d: dict):
    """Python字典 -> Lua表"""
    lua_table = engine.lua.table()
    
    for key, value in d.items():
        # 递归转换嵌套字典

def ipc_loop():
    """IPC主循环"""
    while True:

        if not line:
            break

## 7. 热键注册机制

### 7.1 全局热键注册
```python
import keyboard

class HotkeyManager:
    def __init__(self):
        self.registered = {}
    
    def register(self, hotkey: str, callback: callable) -> str:
        """注册全局热键"""
        hotkey_id = f"hk_{len(self.registered)}"
        
        try:
            keyboard.add_hotkey(hotkey, callback, suppress=True)
            self.registered[hotkey_id] = {
                'hotkey': hotkey,
                'callback': callback
            }
            return hotkey_id
        except Exception as e:
            raise HotkeyError(f"Failed to register hotkey: {e}")
    
    def unregister(self, hotkey_id: str):
        """注销热键"""
        if hotkey_id in self.registered:
            hotkey = self.registered[hotkey_id]['hotkey']
            keyboard.remove_hotkey(hotkey)
            del self.registered[hotkey_id]
    
    def unregister_all(self):
        """注销所有热键"""
        keyboard.unhook_all()
        self.registered.clear()
```

## 8. 数据持久化

### 8.1 原子性写入
```python
import os
import shutil
import json
from threading import Lock

class AtomicWriter:
    def __init__(self, file_path: str):
        self.file_path = file_path
        self.lock = Lock()
    
    def write(self, data: dict):
        """原子性写入JSON"""
        with self.lock:
            temp_path = self.file_path + '.tmp'
            backup_path = self.file_path + '.bak'
            
            try:
                # 1. 写入临时文件
                with open(temp_path, 'w', encoding='utf-8') as f:
                    json.dump(data, f, ensure_ascii=False, indent=2)
                
                # 2. 备份旧文件
                if os.path.exists(self.file_path):
                    shutil.copy2(self.file_path, backup_path)
                
                # 3. 原子替换
                os.replace(temp_path, self.file_path)
                
            except Exception as e:
                # 清理临时文件
                if os.path.exists(temp_path):
                    os.remove(temp_path)
                raise e
```

### 8.2 增量保存
```python
class IncrementalSaver:
    def __init__(self, file_path: str, save_interval: int = 300):
        self.file_path = file_path
        self.save_interval = save_interval
        self.dirty = False
        self.data = self._load()
        self._start_auto_save()
    
    def _load(self) -> dict:
        """加载数据"""
        if os.path.exists(self.file_path):
            with open(self.file_path, 'r', encoding='utf-8') as f:
                return json.load(f)
        return {}
    
    def update(self, key: str, value: any):
        """更新数据"""
        self.data[key] = value
        self.dirty = True
    
    def _start_auto_save(self):
        """启动自动保存线程"""
        import threading
        
        def auto_save_loop():
            while True:
                time.sleep(self.save_interval)
                if self.dirty:
                    self.save()
        
        thread = threading.Thread(target=auto_save_loop, daemon=True)
        thread.start()
    
    def save(self):
        """保存到磁盘"""
        if self.dirty:
            writer = AtomicWriter(self.file_path)
            writer.write(self.data)
            self.dirty = False
```

def migrate_1_0_to_1_1():

### 9.1 PaddleOCR初始化
```python
from paddleocr import PaddleOCR

class OCREngine:
    def __init__(self, use_gpu: bool = False):
        self.ocr = PaddleOCR(
            use_angle_cls=True,  # 支持文字方向检测
            lang='ch',  # 中文模型（支持中英文）
            use_gpu=use_gpu,
            show_log=False
        )
    
    def recognize(self, image: np.ndarray) -> list:
        """识别图像中的文字"""
        result = self.ocr.ocr(image, cls=True)
        
        texts = []
        for line in result:
            for word_info in line:
                bbox = word_info[0]  # [[x1,y1], [x2,y2], [x3,y3], [x4,y4]]
                text = word_info[1][0]  # 文本
                confidence = word_info[1][1]  # 置信度
                
                texts.append({
                    'text': text,
                    'confidence': confidence,
                    'bbox': [{'x': int(p[0]), 'y': int(p[1])} for p in bbox]
                })
        
        return texts
```

### 9.2 OCR结果过滤
```python
def filter_ocr_results(results: list, min_confidence: float = 0.7) -> list:
    """过滤低置信度结果"""
    return [r for r in results if r['confidence'] >= min_confidence]

def find_text_in_results(results: list, target_text: str, 
                         fuzzy: bool = False) -> list:
    """在OCR结果中查找文本"""
    matches = []
    
    for result in results:
        text = result['text']
        
        if fuzzy:
            # 模糊匹配（使用编辑距离）
            from difflib import SequenceMatcher
            ratio = SequenceMatcher(None, text, target_text).ratio()
            if ratio >= 0.8:
                matches.append(result)
        else:
            # 精确匹配
            if target_text in text:
                matches.append(result)
    
    return matches
```

## 10. 性能优化技巧

### 10.1 图像识别缓存
```python
from functools import lru_cache
import hashlib

class RecognitionCache:
    def __init__(self, max_size: int = 100, ttl: int = 1000):
        self.cache = {}
        self.max_size = max_size
        self.ttl = ttl  # 毫秒
    
    def _image_hash(self, image: np.ndarray) -> str:
        """计算图像哈希"""
        return hashlib.md5(image.tobytes()).hexdigest()
    
    def get(self, target_id: str, image: np.ndarray):
        """获取缓存结果"""
        key = f"{target_id}_{self._image_hash(image)}"
        
        if key in self.cache:
            result, timestamp = self.cache[key]
            if time.time() * 1000 - timestamp < self.ttl:
                return result
            else:
                del self.cache[key]
        
        return None
    
    def set(self, target_id: str, image: np.ndarray, result: any):
        """设置缓存"""
        key = f"{target_id}_{self._image_hash(image)}"
        
        # LRU清理
        if len(self.cache) >= self.max_size:
            oldest = min(self.cache.items(), key=lambda x: x[1][1])
            del self.cache[oldest[0]]
        
        self.cache[key] = (result, time.time() * 1000)
```

### 10.3 并行处理
```python
from concurrent.futures import ThreadPoolExecutor, as_completed

class ParallelRecognizer:
    def __init__(self, max_workers: int = 3):
        self.executor = ThreadPoolExecutor(max_workers=max_workers)
    
    def recognize_multiple(self, tasks: list) -> list:
        """并行识别多个目标"""
        futures = {}
        
        for task in tasks:
            future = self.executor.submit(
                self._recognize_one, 
                task['target_id'], 
                task['image']
            )
            futures[future] = task
        
        results = []
        for future in as_completed(futures):
            task = futures[future]
            try:
                result = future.result()
                results.append({
                    'target_id': task['target_id'],
                    'result': result
                })
            except Exception as e:
                results.append({
                    'target_id': task['target_id'],
                    'error': str(e)
                })
        
        return results
```

### 10.4 内存池
```python
class ImageBufferPool:
    def __init__(self, buffer_size: int = 10):
        self.pool = []
        self.buffer_size = buffer_size
    
    def acquire(self, shape: tuple, dtype=np.uint8) -> np.ndarray:
        """获取缓冲区"""
        for i, buf in enumerate(self.pool):
            if buf.shape == shape and buf.dtype == dtype:
                return self.pool.pop(i)
        
        # 没有可用的，创建新的
        return np.empty(shape, dtype=dtype)
    
    def release(self, buffer: np.ndarray):
        """归还缓冲区"""
        if len(self.pool) < self.buffer_size:
            self.pool.append(buffer)
```

## 11. 关键技术决策

### 11.1 为什么选择Tauri而不是Electron
- 更小的安装包（~3MB vs ~100MB）
- 更低的内存占用
- 原生Windows API集成更好
- Rust的安全性保证

### 11.2 为什么选择Python作为后端
- OpenCV、PaddleOCR生态完善
- 快速开发和迭代
- 与Tauri的IPC通信简单
- 后期可以用Cython优化性能

### 11.3 为什么选择Lua作为脚本语言
- 轻量级，易嵌入
- 学习成本低
- 执行效率高
- 沙箱隔离容易实现

### 11.4 为什么使用JSON而不是SQLite（初期）
- 简单直观，易于调试
- 无需额外依赖
- 数据量小时性能足够
- 易于迁移到SQLite
